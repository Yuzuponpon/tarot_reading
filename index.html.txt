  <script>
    // ★ここに16枚分のカード情報を書きます
    const cards = [
      {
        id: "card01",
        title: "戦車",
        keyword: "前進・勝利・勢い",
        image: "cards/card01.png",
        text: "今のあなたには前に進む力があります。少し不安でも、アクセルを踏んで行動してみて。"
      },
      {
        id: "card02",
        title: "隠者",
        keyword: "内省・準備・一人の時間",
        image: "cards/card02.png",
        text: "今日はあえて一人の時間を大切にする日。静かな場所で、自分の本音に耳を澄ませてみて。"
      },
      // ……この要領で 16枚ぶん追加……
      // {
      //   id: "card03",
      //   title: "〜〜",
      //   keyword: "キーワード",
      //   image: "cards/card03.png",
      //   text: "説明文……"
      // },
    ];

    const cardImage = document.getElementById("card-image");
    const startBtn = document.getElementById("start-btn");
    const resultBox = document.getElementById("result");
    const titleEl = document.getElementById("card-title");
    const keywordEl = document.getElementById("card-keyword");
    const textEl = document.getElementById("card-text");

    let timer = null;
    let currentIndex = 0;

    function startRoulette() {
      if (cards.length === 0) return;

      resultBox.classList.add("hidden"); // 前回の結果を隠す
      startBtn.disabled = true;

      // 高速でカードを切り替える（80msごと）
      timer = setInterval(() => {
        currentIndex = Math.floor(Math.random() * cards.length);
        cardImage.src = cards[currentIndex].image;
      }, 80);
    }

    function stopRoulette() {
      if (!timer) return; // 動いていなければ何もしない

      clearInterval(timer);
      timer = null;

      const card = cards[currentIndex];
      titleEl.textContent = card.title;
      keywordEl.textContent = card.keyword;
      textEl.textContent = card.text;

      resultBox.classList.remove("hidden");
      startBtn.disabled = false;
    }

    // ボタンでスタート
    startBtn.addEventListener("click", startRoulette);

    // 画像タップでストップ
    cardImage.addEventListener("click", stopRoulette);
  </script>
